  1014→function LeadForm() {
  1015→  const [form, setForm] = useState<LeadState>({ name: "", email: "", business: "", industry: "", budget: "", brief: "" })
  1016→  const [errors, setErrors] = useState<{ name?: string; email?: string }>({})
  1017→  const [status, setStatus] = useState<"idle" | "sending" | "done" | "error">("idle")
  1018→  const [refCode, setRefCode] = useState("")
  1019→  const started = useRef(false)
  1020→
  1021→  const bind = (key: keyof LeadState) => ({
  1022→    value: form[key],
  1023→    onChange: (e: ChangeEvent<HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement>) =>
  1024→      setForm((f) => ({ ...f, [key]: e.target.value })),
  1025→  })
  1026→

  1035→    e.preventDefault()
  1036→    const errs: typeof errors = {}

  1038→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(form.email.trim())) errs.email = "A complete email, so we can reply."

  1077→    )
  1078→  }
  1079→
