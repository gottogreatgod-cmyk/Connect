   193→/** session persistence — the build survives reloads; heavy results
   194→ * (base64 images, video urls, audio) are dropped, prompts and picks stay */
   195→const SS_KEY = "gos:builder"

   197→function persist(s: BuilderState) {
   198→  if (typeof window === "undefined" || typeof sessionStorage === "undefined") return
   199→  try {
   200→    const light: BuilderState = {
   201→      ...s,
   202→      copyBusy: false,
   203→      copyError: null,
   204→      refBusy: false,
   205→      refError: null,
   206→      refNote: null,
   207→      assets: Object.fromEntries(
   208→        Object.entries(s.assets).map(([k, a]) => [
   209→          k,
   210→          {
   211→            ...a,
   212→            image: undefined,
   213→            video: undefined,
   214→            audio: undefined,
   215→            status: a.status === "running" || a.status === "done" ? "idle" : a.status,
   216→            startedAt: undefined,
   217→          },
   218→        ]),
   219→      ) as Record<AssetId, AssetState>,
   220→    }
   221→    sessionStorage.setItem(SS_KEY, JSON.stringify(light))
   222→  } catch {
   223→    /* quota exceeded or storage blocked — the in-memory build still works */
   224→  }
   225→}

   227→function loadPersisted(): BuilderState {
   228→  const fallback: BuilderState = {
   229→    brief: "",
   230→    tplSlug: null,
   231→    edition: null,
   232→    genome: { ...NATIVE },
   233→    name: "",
   234→    city: "",
   235→    pitch: "",
   236→    persona: "warm",
   237→    copy: null,
   238→    copyRaw: "",
   239→    copyBusy: false,
   240→    copyError: null,
   241→    refBusy: false,
   242→    refError: null,
   243→    refNote: null,
   244→    personality: null,
   245→    assets: initialAssets(null),
   246→  }
   247→  try {
   248→    if (typeof window === "undefined" || typeof sessionStorage === "undefined") return fallback
   249→    const raw = sessionStorage.getItem(SS_KEY)
   250→    if (!raw) return fallback
   251→    const parsed = JSON.parse(raw) as BuilderState
   252→    if (!parsed || typeof parsed !== "object" || !parsed.assets) return fallback
   253→    /* genome shape guard — a stale or foreign payload falls back cleanly */
   254→    if (!parsed.genome || typeof parsed.genome !== "object") parsed.genome = { ...NATIVE }
   255→    return { ...fallback, ...parsed }
   256→  } catch {
   257→    return fallback
   258→  }
   259→}

   261→let state: BuilderState = loadPersisted()
   262→
   263→const listeners = new Set<() => void>()
   264→
   265→function setState(patch: Partial<BuilderState>) {
   266→  state = { ...state, ...patch }
   267→  persist(state)
   268→  listeners.forEach((l) => l())
   269→}
   270→
   271→function setAsset(id: AssetId, patch: Partial<AssetState>) {
   272→  setState({ assets: { ...state.assets, [id]: { ...state.assets[id], ...patch } } })
   273→}
   274→
   275→function subscribe(l: () => void) {
   276→  listeners.add(l)
   277→  return () => {
   278→    listeners.delete(l)
   279→  }
   280→}
   281→
   282→function useBuilder(): BuilderState {
   283→  return useSyncExternalStore(subscribe, () => state, () => state)
   284→}

   288→async function generateImage(id: "hero" | "mark" | "still") {
   289→  const a = state.assets[id]
   290→  if (!a.prompt.trim() || a.status === "running") return
   291→  track("ai_skill_used", `builder/image-${id}`)
   292→  setAsset(id, { status: "running", error: undefined, image: undefined, startedAt: Date.now() })
   293→  try {
   294→    const { jobId } = await postJson<{ jobId: string }>("/api/ai/image", { prompt: a.prompt, size: a.size })
   295→    await pollJob("image", jobId, (s: PollState) => {
   296→      if (s.status === "done") setAsset(id, { status: "done", image: s.result?.image })
   297→      else if (s.status === "error") setAsset(id, { status: "error", error: s.error })
   298→      else setAsset(id, { status: "running" })
   299→    })
   300→  } catch (e) {
   301→    setAsset(id, { status: "error", error: e instanceof Error ? e.message : "Could not start the render." })
   302→  }
   303→}

   351→/** The reference-analysis entry point (the openmontage pattern): paste
   352→ * any site URL, the Page Reader extracts it, the chat AI distills a
   353→ * one-line brief, and the recommendations re-rank against it. */
   354→async function analyzeReference(url: string) {
   355→  if (!url.trim() || state.refBusy) return
   356→  track("ai_skill_used", "builder/reference")
   357→  setState({ refBusy: true, refError: null, refNote: null })
   358→  try {
   359→    const read = await postJson<{ text?: string; title?: string; error?: string }>("/api/ai/reader", { url })
   360→    const text = (read.text ?? "").slice(0, 5000)
   361→    if (!text) throw new Error(read.error ?? "That page returned no readable content.")
   362→    const res = await postJson<{ reply?: string; error?: string }>("/api/ai/chat", {
   363→      system:
   364→        "You turn a website's readable text into a one-line business brief for a website builder. " +
   365→        "Reply with EXACTLY one line, max 20 words, in the shape: 'a <what they do> in <city if known> that <conversion goal>'. " +
   366→        "No quotes, no preamble, no explanation — just the line.",
   367→      messages: [{ role: "user", content: `Website: ${read.title ?? url}\n\nPAGE TEXT:\n${text}` }],
   368→    })
   369→    const line = (res.reply ?? "").replace(/^["']|["']$/g, "").trim()
   370→    if (!line) throw new Error(res.error ?? "Could not distill that page.")
   371→    setState({ brief: line, refBusy: false, refNote: `From ${read.title ?? url} — edit freely, then pick a starting point.` })
   372→  } catch (e) {
   373→    setState({ refBusy: false, refError: e instanceof Error ? e.message : "Could not read that reference." })
   374→  }
   375→}

   377→async function writeCopy() {
   378→  if (state.copyBusy) return
   379→  const t = state.tplSlug ? findTemplate(state.tplSlug) : null
   380→  if (!t) return
   381→  track("ai_skill_used", "builder/copy")
   382→  setState({ copyBusy: true, copyError: null })
   383→  const business = state.name.trim() || t.name
   384→  const city = state.city.trim() || t.city
   385→  const pitch = state.pitch.trim()
   386→  const persona = PERSONAS.find((p) => p.id === state.persona) ?? PERSONAS[0]
   387→  try {
   388→    const res = await postJson<{ reply?: string; error?: string }>("/api/ai/chat", {
   389→      system:
   390→        `You write launch copy for small-business websites. ${persona.style}\n` +
   391→        `Grounded context about the reference template:\n${templateDigest(t)}\n` +
   392→        `The client's business: name "${business}", city ${city}${pitch ? `, pitch: ${pitch}` : ""}.\n` +
   393→        `Write fresh launch copy for THE CLIENT'S business (not the template's demo brand). Keep it concrete and honest, no hype words like "unlock" or "elevate".\n` +
   394→        `Answer with EXACTLY four lines, each starting with the label, nothing else:\n` +
   395→        `HEADLINE: <max 8 words>\nTAGLINE: <one sentence>\nCTA: <max 4 words, a button label>\nABOUT: <2 sentences>`,
   396→      messages: [{ role: "user", content: `Write the launch copy for ${business} in ${city}.` }],
   397→    })
   398→    const raw = res.reply ?? ""
   399→    if (!raw) throw new Error(res.error ?? "No copy came back.")
   400→    const pick = (label: string) => {
   401→      const m = raw.match(new RegExp(`^\\s*${label}\\s*:\\s*(.+)$`, "im"))
   402→      return m ? m[1].trim() : ""
   403→    }
   404→    const block: CopyBlock = {
   405→      headline: pick("HEADLINE"),
   406→      tagline: pick("TAGLINE"),
   407→      cta: pick("CTA"),
   408→      about: pick("ABOUT"),
   409→    }
   410→    if (!block.headline && !block.tagline && !block.about) {
   411→      /* model ignored the format — keep the raw text visible, still useful */
   412→      setState({ copyRaw: raw, copy: null, copyBusy: false })
   413→      return
   414→    }
   415→    setState({ copy: block, copyRaw: raw, copyBusy: false })
   416→    /* re-seed untouched asset prompts with the client's OWN business —
   417→       the template's demo brand never survives into the art step */
   418→    const pos = state.pitch.trim() || t.positioning.split("—")[0].trim()
   419→    const assets = { ...state.assets }
   420→    if (assets.hero.status === "idle" && !assets.hero.touched)
   421→      assets.hero = { ...assets.hero, prompt: `${pos}, ${city}, hero website background, soft natural light, photographic, no text` }
   422→    if (assets.mark.status === "idle" && !assets.mark.touched)
   423→      assets.mark = { ...assets.mark, prompt: `minimal monogram for a business called "${business}", premium, flat vector, centered on plain background, no text effects` }
   424→    if (assets.still.status === "idle" && !assets.still.touched)
   425→      assets.still = { ...assets.still, prompt: `editorial detail photograph — ${pos}, ${city}, shallow depth of field, magazine quality` }
   426→    if (assets.reel.status === "idle" && !assets.reel.touched)
   427→      assets.reel = { ...assets.reel, prompt: `cinematic slow camera move, ${pos.toLowerCase()}, ${city}, warm light, premium mood, no text overlays` }
   428→    setState({ assets })
   429→  } catch (e) {
   430→    setState({ copyBusy: false, copyError: e instanceof Error ? e.message : "The copy service is briefly unavailable." })
   431→  }
   432→}

   570→/* ── the "next up" recommendation card ────────────────────────────── */

   572→function nextUpOf(s: BuilderState): { step: BuilderStepId; label: string; why: string } {
   573→  if (!s.tplSlug) return { step: "brief", label: "Describe what you're building", why: "One sentence is enough — you'll get honest template picks." }
