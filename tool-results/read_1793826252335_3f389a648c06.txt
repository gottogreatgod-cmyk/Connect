   153→   153→function useMediaList(kind: "image" | "audio" | "video") {
   154→   154→  const [items, setItems] = useState<MediaItem[]>([])
   155→   155→  const load = useCallback(async () => {
   156→   156→    try {
   157→   157→      const res = await fetch(`/api/media?kind=${kind}&take=6`)
   158→   158→      if (res.ok) setItems((await res.json()).media)
   159→   159→    } catch {
   160→   160→      /* non-fatal */
   161→   161→    }
   162→   162→  }, [kind])
   163→   163→  useEffect(() => {
   164→   164→    const t = setTimeout(load, 0)
   165→   165→    return () => clearTimeout(t)
   166→   166→  }, [load])
   167→   167→  return { items, reload: load }
   168→   168→}

   186→   186→function AsrTool({ onSendToChat }: { onSendToChat?: (text: string) => void }) {
   187→   187→  const { toast } = useToast()
   188→   188→  const [recording, setRecording] = useState(false)
   189→   189→  const [seconds, setSeconds] = useState(0)
   190→   190→  const [busy, setBusy] = useState(false)
   191→   191→  const [transcript, setTranscript] = useState("")
   192→   192→  const [copied, setCopied] = useState(false)
   193→   193→  const recorderRef = useRef<MediaRecorder | null>(null)
   194→   194→  const chunksRef = useRef<Blob[]>([])
   195→   195→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   196→   196→  const fileRef = useRef<HTMLInputElement>(null)
   197→   197→
   198→   198→  const transcribe = useCallback(
   199→   199→    async (base64: string) => {
   200→   200→      setBusy(true)
   201→   201→      try {
   202→   202→        const res = await fetch("/api/media/asr", {
   203→   203→          method: "POST",
   204→   204→          headers: { "content-type": "application/json" },
   205→   205→          body: JSON.stringify({ audioBase64: base64 }),
   206→   206→        })
   207→   207→        const data = await res.json()
   208→   208→        if (!res.ok) throw new Error(data.error ?? "Transcription failed")
   209→   209→        setTranscript(data.text)
   210→   210→      } catch (err) {
   211→   211→        toast({ title: "Transcription failed", description: err instanceof Error ? err.message : String(err), variant: "destructive" })
   212→   212→      } finally {
   213→   213→        setBusy(false)
   214→   214→      }
   215→   215→    },
   216→   216→    [toast],
   217→   217→  )
   218→   218→
   219→   219→  const stopAndTranscribe = useCallback(async () => {
   220→   220→    const recorder = recorderRef.current
   221→   221→    if (!recorder) return
   222→   222→    recorder.stop()
   223→   223→    recorder.stream.getTracks().forEach((t) => t.stop())
   224→   224→    if (timerRef.current) clearInterval(timerRef.current)
   225→   225→    setRecording(false)
   226→   226→    const blob = new Blob(chunksRef.current, { type: recorder.mimeType || "audio/webm" })
   227→   227→    try {
   228→   228→      const base64 = await blobToWavBase64(blob)
   229→   229→      await transcribe(base64)
   230→   230→    } catch (err) {
   231→   231→      toast({ title: "Could not process the recording", description: err instanceof Error ? err.message : String(err), variant: "destructive" })
   232→   232→      setBusy(false)
   233→   233→    }
   234→   234→  }, [transcribe, toast])
   235→   235→
   236→   236→  const startRecording = useCallback(async () => {
   237→   237→    try {
   238→   238→      const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
   239→   239→      const recorder = new MediaRecorder(stream)
   240→   240→      chunksRef.current = []
   241→   241→      recorder.ondataavailable = (e) => {
   242→   242→        if (e.data.size > 0) chunksRef.current.push(e.data)
   243→   243→      }
   244→   244→      recorder.start()
   245→   245→      recorderRef.current = recorder
   246→   246→      setRecording(true)
   247→   247→      setSeconds(0)
   248→   248→      setTranscript("")
   249→   249→      timerRef.current = setInterval(() => setSeconds((s) => s + 1), 1000)
   250→   250→    } catch {
   251→   251→      toast({ title: "Microphone unavailable", description: "Grant mic permission, or upload an audio file instead.", variant: "destructive" })
   252→   252→    }
   253→   253→  }, [toast])
   254→   254→
   255→   255→  useEffect(
   256→   256→    () => () => {
   257→   257→      if (timerRef.current) clearInterval(timerRef.current)
   258→   258→      recorderRef.current?.stream.getTracks().forEach((t) => t.stop())
   259→   259→    },
   260→   260→    [],
   261→   261→  )
   262→   262→
   263→   263→  const onFile = async (file: File) => {
   264→   264→    const ext = file.name.split(".").pop()?.toLowerCase() ?? ""
   265→   265→    setBusy(true)
   266→   266→    setTranscript("")
   267→   267→    try {
   268→   268→      const base64 = /^(wav|mp3|m4a|flac|ogg)$/.test(ext)
   269→   269→        ? await fileToBase64(file)
   270→   270→        : await blobToWavBase64(file)
   271→   271→      await transcribe(base64)
   272→   272→    } catch (err) {
   273→   273→      toast({ title: "Could not read that file", description: err instanceof Error ? err.message : String(err), variant: "destructive" })
   274→   274→      setBusy(false)
   275→   275→    }
   276→   276→  }
   277→   277→
   278→   278→  return (
   279→   279→    <div className="rounded-xl border border-border/70 bg-card/40 p-4">
   280→   280→      <ToolHeader icon={<AudioLines className="h-4 w-4" />} title="Voice → Text" note="16 kHz mono capture · exact transcription" />
   281→   281→
   282→   282→      <div className="flex flex-col items-center gap-4 py-4">
   283→   283→        <button
   284→   284→          onClick={recording ? stopAndTranscribe : startRecording}
   285→   285→          disabled={busy}
   286→   286→          aria-label={recording ? "Stop recording" : "Start recording"}
   287→   287→          className={cn(
   288→   288→            "relative flex h-20 w-20 items-center justify-center rounded-full border transition-all",
   289→   289→            recording
   290→   290→              ? "border-primary/60 bg-primary/20 gos-glow"
   291→   291→              : "border-primary/30 bg-primary/10 hover:bg-primary/20",
   292→   292→            busy && "opacity-60",
   293→   293→          )}
   294→   294→        >
   295→   295→          {busy ? (
   296→   296→            <Loader2 className="h-7 w-7 animate-spin text-primary" />
   297→   297→          ) : recording ? (
   298→   298→            <Square className="h-6 w-6 text-primary" />
   299→   299→          ) : (
   300→   300→            <Mic className="h-7 w-7 text-primary" />
   301→   301→          )}
   302→   302→          {recording && <span className="gos-pulse-dot absolute inset-0 rounded-full border-2 border-primary/50" />}
   303→   303→        </button>
   304→   304→        <div className="text-center">
   305→   305→          <p className="font-mono text-sm tabular-nums text-primary">
   306→   306→            {recording
   307→   307→              ? `recording · ${Math.floor(seconds / 60)}:${String(seconds % 60).padStart(2, "0")}`
   308→   308→              : busy
   309→   309→                ? "transcribing…"
   310→   310→                : "tap to speak"}
   311→   311→          </p>
   312→   312→          <p className="mt-1 font-mono text-[10px] text-muted-foreground">
   313→   313→            or upload a file — wav · mp3 · m4a · flac · ogg
   314→   314→          </p>
   315→   315→        </div>
   316→   316→        <input
   317→   317→          ref={fileRef}
   318→   318→          type="file"
   319→   319→          accept="audio/*"
   320→   320→          className="hidden"
   321→   321→          onChange={(e) => {
   322→   322→            const f = e.target.files?.[0]
   323→   323→            if (f) onFile(f)
   324→   324→            e.target.value = ""
   325→   325→          }}
   326→   326→        />
   327→   327→        <Button variant="outline" size="sm" className="gap-2" onClick={() => fileRef.current?.click()} disabled={recording || busy}>
   328→   328→          <Upload className="h-3.5 w-3.5" />
   329→   329→          Upload audio
   330→   330→        </Button>
   331→   331→      </div>
   332→   332→
   333→   333→      {transcript && (
   334→   334→        <div className="gos-rise rounded-xl border border-primary/25 bg-accent/30 p-4">
   335→   335→          <p className="whitespace-pre-wrap text-[15px] leading-relaxed">{transcript}</p>
   336→   336→          <div className="mt-3 flex flex-wrap gap-2">
   337→   337→            <Button
   338→   338→              variant="outline"
   339→   339→              size="sm"
   340→   340→              className="gap-1.5"
   341→   341→              onClick={() => {
   342→   342→                void navigator.clipboard.writeText(transcript)
   343→   343→                setCopied(true)
   344→   344→                setTimeout(() => setCopied(false), 1500)
   345→   345→              }}
   346→   346→            >
   347→   347→              {copied ? <Check className="h-3.5 w-3.5 text-primary" /> : <Copy className="h-3.5 w-3.5" />}
   348→   348→              {copied ? "Copied" : "Copy"}
   349→   349→            </Button>
   350→   350→            {onSendToChat && (
   351→   351→              <Button size="sm" className="gap-1.5" onClick={() => onSendToChat(transcript)}>
   352→   352→                <Sparkles className="h-3.5 w-3.5" />
   353→   353→                Run through the agent loop
   354→   354→              </Button>
   355→   355→            )}
   356→   356→            <span className="ml-auto self-center font-mono text-[10px] text-muted-foreground">
   357→   357→              {transcript.split(/\s+/).filter(Boolean).length} words
   358→   358→            </span>
   359→   359→          </div>
   360→   360→        </div>
   361→   361→      )}
   362→   362→    </div>
   363→   363→  )
   364→   364→}

   368→   368→function TtsTool() {
   369→   369→  const { toast } = useToast()
   370→   370→  const [text, setText] = useState("")
   371→   371→  const [voice, setVoice] = useState("tongtong")
   372→   372→  const [speed, setSpeed] = useState(1)
   373→   373→  const [busy, setBusy] = useState(false)
   374→   374→  const [audioUrl, setAudioUrl] = useState<string | null>(null)
   375→   375→  const [audioId, setAudioId] = useState<string | null>(null)
   376→   376→  const { items, reload } = useMediaList("audio")
   377→   377→  const over = text.length > 1024
   378→   378→
   379→   379→  const generate = async () => {
   380→   380→    if (!text.trim() || over || busy) return
   381→   381→    setBusy(true)
   382→   382→    try {
   383→   383→      const res = await fetch("/api/media/tts", {
   384→   384→        method: "POST",
   385→   385→        headers: { "content-type": "application/json" },
   386→   386→        body: JSON.stringify({ text, voice, speed }),
   387→   387→      })
   388→   388→      if (!res.ok) {
   389→   389→        const data = await res.json().catch(() => ({ error: "Synthesis failed" }))
   390→   390→        throw new Error(data.error ?? "Synthesis failed")
   391→   391→      }
   392→   392→      const blob = await res.blob()
   393→   393→      setAudioUrl(URL.createObjectURL(blob))
   394→   394→      setAudioId(res.headers.get("X-Media-Id"))
   395→   395→      reload()
   396→   396→    } catch (err) {
   397→   397→      toast({ title: "Voice synthesis failed", description: err instanceof Error ? err.message : String(err), variant: "destructive" })
   398→   398→    } finally {
   399→   399→      setBusy(false)
   400→   400→    }
   401→   401→  }
   402→   402→
   403→   403→  return (
   404→   404→    <div className="rounded-xl border border-border/70 bg-card/40 p-4">
   405→   405→      <ToolHeader icon={<Volume2 className="h-4 w-4" />} title="Text → Voice" note="1024 characters per track · 7 voices" />
   406→   406→
   407→   407→      <Textarea
   408→   408→        value={text}
   409→   409→        onChange={(e) => setText(e.target.value)}
   410→   410→        placeholder="Type anything — a greeting, a narration, a line for your site's hero…"
   411→   411→        rows={4}
   412→   412→        className="resize-none text-[14px] leading-relaxed"
   413→   413→      />
   414→   414→      <div className={cn("mt-1 text-right font-mono text-[10px]", over ? "text-destructive" : "text-muted-foreground")}>
   415→   415→        {text.length}/1024
   416→   416→      </div>
   417→   417→
   418→   418→      <div className="mt-3 grid gap-3 sm:grid-cols-2">
   419→   419→        <div>
   420→   420→          <label htmlFor="tts-voice" className="mb-1 block font-mono text-[10px] uppercase tracking-widest text-muted-foreground">
   421→   421→            voice
   422→   422→          </label>
   423→   423→          <Select value={voice} onValueChange={setVoice}>
   424→   424→            <SelectTrigger id="tts-voice" className="w-full">
   425→   425→              <SelectValue />
   426→   426→            </SelectTrigger>
   427→   427→            <SelectContent>
   428→   428→              {VOICES.map((v) => (
   429→   429→                <SelectItem key={v.id} value={v.id}>
   430→   430→                  {v.label} <span className="font-mono text-[10px] text-muted-foreground">· {v.id}</span>
   431→   431→                </SelectItem>
   432→   432→              ))}
   433→   433→            </SelectContent>

   569→   569→function VideoTool() {
   570→   570→  const { toast } = useToast()
   571→   571→  const [prompt, setPrompt] = useState("")
   572→   572→  const [quality, setQuality] = useState("speed")
   573→   573→  const [busy, setBusy] = useState(false)
   574→   574→  const [rendering, setRendering] = useState<{ taskId: string; mediaId: string; prompt: string } | null>(null)
   575→   575→  const [elapsed, setElapsed] = useState(0)
   576→   576→  const [result, setResult] = useState<MediaRef | null>(null)
   577→   577→  const { items, reload } = useMediaList("video")
   578→   578→  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null)
   579→   579→  const tickRef = useRef<ReturnType<typeof setInterval> | null>(null)
   580→   580→  const startRef = useRef(0)
   581→   581→
   582→   582→  const stopTimers = () => {
   583→   583→    if (pollRef.current) clearInterval(pollRef.current)
   584→   584→    if (tickRef.current) clearInterval(tickRef.current)
   585→   585→    pollRef.current = null
   586→   586→    tickRef.current = null
   587→   587→  }
   588→   588→
   589→   589→  useEffect(() => stopTimers, [])
   590→   590→
   591→   591→  const start = async () => {
   592→   592→    if (!prompt.trim() || busy || rendering) return
   593→   593→    setBusy(true)
   594→   594→    setResult(null)
   595→   595→    try {
   596→   596→      const res = await fetch("/api/media/video", {
   597→   597→        method: "POST",
   598→   598→        headers: { "content-type": "application/json" },
   599→   599→        body: JSON.stringify({ prompt, quality }),
   600→   600→      })
   601→   601→      const data = await res.json()
   602→   602→      if (!res.ok) throw new Error(data.error ?? "Could not start the render")
   603→   603→      setBusy(false)
   604→   604→      setRendering({ taskId: data.taskId, mediaId: data.mediaId, prompt })
   605→   605→      startRef.current = Date.now()
   606→   606→      setElapsed(0)
   607→   607→      tickRef.current = setInterval(() => setElapsed(Math.floor((Date.now() - startRef.current) / 1000)), 1000)
   608→   608→      pollRef.current = setInterval(
   609→   609→        async () => {
   610→   610→          try {
   611→   611→            const sres = await fetch(`/api/media/video/status?taskId=${data.taskId}`)
   612→   612→            if (!sres.ok) return
   613→   613→            const sdata = await sres.json()
   614→   614→            if (sdata.status === "SUCCESS" && sdata.url) {
   615→   615→              stopTimers()
   616→   616→              setRendering(null)
   617→   617→              setResult({ id: data.mediaId, kind: "video", prompt, url: sdata.url })
   618→   618→              reload()
   619→   619→            } else if (sdata.status === "FAIL") {
   620→   620→              stopTimers()
   621→   621→              setRendering(null)
   622→   622→              toast({ title: "Render failed", description: "The render farm reported a failure — try a simpler scene.", variant: "destructive" })
   623→   623→            }
   624→   624→          } catch {
   625→   625→            /* keep polling */
   626→   626→          }
   627→   627→        },
   628→   628→        5000,
   629→   629→      )
   630→   630→    } catch (err) {
   631→   631→      toast({ title: "Could not start the render", description: err instanceof Error ? err.message : String(err), variant: "destructive" })
   632→   632→      setBusy(false)
   633→   633→    }
   634→   634→  }
   635→   635→
   636→   636→  return (
   637→   637→    <div className="rounded-xl border border-border/70 bg-card/40 p-4">
   638→   638→      <ToolHeader icon={<Clapperboard className="h-4 w-4" />} title="Video" note="async render farm · live progress · mp4 with audio" />
   639→   639→
   640→   640→      <Textarea
   641→   641→        value={prompt}
   642→   642→        onChange={(e) => setPrompt(e.target.value)}
   643→   643→        placeholder="A slow dolly shot through a sunlit ceramic studio, dust in the air, warm tones"
   644→   644→        rows={3}
   645→   645→        className="resize-none text-[14px] leading-relaxed"
   646→   646→      />
   647→   647→
   648→   648→      <div className="mt-3">
   649→   649→        <label htmlFor="vid-quality" className="mb-1 block font-mono text-[10px] uppercase tracking-widest text-muted-foreground">
   650→   650→          render mode
   651→   651→        </label>
   652→   652→        <Select value={quality} onValueChange={setQuality}>
   653→   653→          <SelectTrigger id="vid-quality" className="w-full sm:w-72">
   654→   654→            <SelectValue />
   655→   655→          </SelectTrigger>
   656→   656→          <SelectContent>
   657→   657→            <SelectItem value="speed">Speed — faster draft</SelectItem>
   658→   658→            <SelectItem value="quality">Quality — slower, finer detail</SelectItem>
   659→   659→          </SelectContent>
   660→   660→        </Select>
   661→   661→      </div>
   662→   662→
   663→   663→      <Button onClick={start} disabled={!prompt.trim() || busy || !!rendering} className="mt-4 w-full gap-2">
