   153→   153→function persistMessages(msgs: AgentMsg[]) {
   154→   154→  if (typeof window === "undefined" || typeof sessionStorage === "undefined") return;
   155→   155→  try {

   158→   158→      text: m.text.slice(0, 4000),
   159→   159→      /* site actions re-stage on restore (small, idempotent); render
   160→   160→         jobs come back as descriptors — never silently re-run */
   161→   161→      actions: m.actions?.filter((a) => a.type === "site"),
   162→   162→      restoredActions: m.actions
   163→   163→        ?.filter((a) => a.type !== "site")
   164→   164→        .map((a) => ({ action: a, restored: true })),
   165→   165→      trace: m.trace?.slice(0, 10),
   166→   166→      memory: m.memory?.slice(0, 8),
   167→   167→    }));
   168→   168→    sessionStorage.setItem(CHAT_KEY, JSON.stringify(light));
   169→   169→  } catch {
   170→   170→    /* quota or storage blocked — the in-memory conversation still works */
   171→   171→  }

   330→   330→/** run-once launcher with cached results: writes to RESULTS always and
   331→   331→ *  to whichever component instance is currently mounted for the rid —
   332→   332→ *  so a render started before navigation keeps updating the view that
   333→   333→ *  remounts after it. */
   334→   334→function useLaunched(ridv: string, run: (write: (r: ActionResult) => void) => Promise<void>): ActionResult {
   335→   335→  const [state, setState] = useState<ActionResult>(() => RESULTS.get(ridv) ?? { phase: "running" });
   336→   336→
   337→   337→  useEffect(() => {
   338→   338→    /* this instance is now the live writer target for the rid */
   339→   339→    LIVE.set(ridv, setState);
   340→   340→    if (!EXECUTED.has(ridv)) {
   341→   341→      EXECUTED.add(ridv);
   342→   342→      void run((r) => {
   343→   343→        RESULTS.set(ridv, r);
   344→   344→        LIVE.get(ridv)?.(r);
   345→   345→      });
   346→   346→    } else {
   347→   347→      /* remount of a running/finished action — push the latest cached
   348→   348→         value through the subscription path (async, never a sync
   349→   349→         setState in the effect body); identical refs bail out in React,
   350→   350→         and later updates flow through LIVE as the job keeps writing */
   351→   351→      const cached = RESULTS.get(ridv);
   352→   352→      if (cached) queueMicrotask(() => LIVE.get(ridv)?.(cached));
   353→   353→    }
   354→   354→    return () => {
   355→   355→      if (LIVE.get(ridv) === setState) LIVE.delete(ridv);
   356→   356→    };
   357→   357→  }, [ridv, run]);
   358→   358→
   359→   359→  return state;
   360→   360→}
