   193→/** session persistence — the build survives reloads; heavy results
   194→ * (base64 images, video urls, audio) are dropped, prompts and picks stay */
   195→const SS_KEY = "gos:builder"
   196→
   197→function persist(s: BuilderState) {
   198→  if (typeof window === "undefined" || typeof sessionStorage === "undefined") return
   199→  try {
   200→    const light: BuilderState = {
   201→      ...s,
   202→      copyBusy: false,
   203→      copyError: null,
   204→      refBusy: false,
   205→      refError: null,
   206→      refNote: null,
   207→      assets: Object.fromEntries(
   208→        Object.entries(s.assets).map(([k, a]) => [
   209→          k,
   210→          {
   211→            ...a,
   212→            image: undefined,
   213→            video: undefined,
   214→            audio: undefined,
   215→            status: a.status === "running" || a.status === "done" ? "idle" : a.status,
   216→            startedAt: undefined,
   217→          },
   218→        ]),
   219→      ) as Record<AssetId, AssetState>,
   220→    }
   221→    sessionStorage.setItem(SS_KEY, JSON.stringify(light))
   222→  } catch {
   223→    /* quota exceeded or storage blocked — the in-memory build still works */
   224→  }
   225→}
   226→
   227→function loadPersisted(): BuilderState {
   228→  const fallback: BuilderState = {
   229→    brief: "",
   230→    tplSlug: null,
   231→    edition: null,
   232→    genome: { ...NATIVE },
   233→    name: "",
   234→    city: "",
   235→    pitch: "",
   236→    persona: "warm",
   237→    copy: null,
   238→    copyRaw: "",
   239→    copyBusy: false,
   240→    copyError: null,
   241→    refBusy: false,
   242→    refError: null,
   243→    refNote: null,
   244→    personality: null,
   245→    assets: initialAssets(null),
   246→  }
   247→  try {
   248→    if (typeof window === "undefined" || typeof sessionStorage === "undefined") return fallback
   249→    const raw = sessionStorage.getItem(SS_KEY)
   250→    if (!raw) return fallback
   251→    const parsed = JSON.parse(raw) as BuilderState
   252→    if (!parsed || typeof parsed !== "object" || !parsed.assets) return fallback
   253→    /* genome shape guard — a stale or foreign payload falls back cleanly */
   254→    if (!parsed.genome || typeof parsed.genome !== "object") parsed.genome = { ...NATIVE }
   255→    return { ...fallback, ...parsed }
   256→  } catch {
   257→    return fallback
   258→  }
   259→}
   260→
   261→let state: BuilderState = loadPersisted()
   262→
   263→const listeners = new Set<() => void>()
   264→
   265→function setState(patch: Partial<BuilderState>) {
   266→  state = { ...state, ...patch }
   267→  persist(state)
   268→  listeners.forEach((l) => l())
   269→}
   270→
   271→function setAsset(id: AssetId, patch: Partial<AssetState>) {
   272→  setState({ assets: { ...state.assets, [id]: { ...state.assets[id], ...patch } } })
   273→}
   274→
   275→function subscribe(l: () => void) {
   276→  listeners.add(l)
   277→  return () => {
   278→    listeners.delete(l)
   279→  }
   280→}

   290→async function generateImage(id: "hero" | "mark" | "still") {
   291→  const a = state.assets[id]
   292→  if (!a.prompt.trim() || a.status === "running") return
   293→  track("ai_skill_used", `builder/image-${id}`)
   294→  setAsset(id, { status: "running", error: undefined, image: undefined, startedAt: Date.now() })
   295→  try {
   296→    const { jobId } = await postJson<{ jobId: string }>("/api/ai/image", { prompt: a.prompt, size: a.size })
   297→    await pollJob("image", jobId, (s: PollState) => {
   298→      if (s.status === "done") setAsset(id, { status: "done", image: s.result?.image })
   299→      else if (s.status === "error") setAsset(id, { status: "error", error: s.error })
   300→      else setAsset(id, { status: "running" })
   301→    })
   302→  } catch (e) {
   303→    setAsset(id, { status: "error", error: e instanceof Error ? e.message : "Could not start the render." })
   304→  }
   305→}

   370→/**
   371→ * Stage a full build from an agent pick — template, launch copy, motion
   372→ * personality and re-seeded asset prompts, all in one write. Used by the
   373→ * Builder's "Generate my website" button AND the Agent chat; the two
   374→ * doors share one store, so whatever either one stages, the other sees.
   375→ */
   376→export function applyAgentBuild(pick: AgentSitePick): TemplateDef | null {
   377→  const t = findTemplate(pick.slug)
   378→  if (!t) return null
   379→  const name = (pick.name ?? "").trim() || ""
   380→  const city = (pick.city ?? "").trim() || t.city
   381→  const pos = (pick.pitch ?? "").trim() || t.positioning.split("—")[0].trim()
   382→  const copy: CopyBlock | null =
   383→    pick.headline || pick.tagline || pick.about
   384→      ? {
   385→          headline: pick.headline ?? "",
   386→          tagline: pick.tagline ?? "",
   387→          cta: pick.cta ?? "",
   388→          about: pick.about ?? "",
   389→        }
   390→      : null
   391→  /* asset prompts seeded from the CLIENT'S business, never the demo brand */
   392→  const assets = initialAssets(t)
   393→  if (name || city) {
   394→    if (!assets.hero.touched) assets.hero = { ...assets.hero, prompt: `${pos}, ${city}, hero website background, soft natural light, photographic, no text` }
   395→    if (!assets.mark.touched) assets.mark = { ...assets.mark, prompt: `minimal monogram for a business called "${name || t.name}", premium, flat vector, centered on plain background, no text effects` }
   396→    if (!assets.still.touched) assets.still = { ...assets.still, prompt: `editorial detail photograph — ${pos}, ${city}, shallow depth of field, magazine quality` }
   397→    if (!assets.reel.touched) assets.reel = { ...assets.reel, prompt: `cinematic slow camera move, ${pos.toLowerCase()}, ${city}, warm light, premium mood, no text overlays` }
   398→  }
   399→  setState({
   400→    brief: (pick.brief ?? "").trim() || state.brief,
   401→    tplSlug: t.slug,
   402→    genome: { ...NATIVE },
   403→    edition: null,
   404→    name,
   405→    city,
   406→    pitch: (pick.pitch ?? "").trim(),
   407→    copy,
   408→    copyRaw: "",
   409→    personality: pick.motion && pick.motion in PERSONALITIES ? pick.motion : state.personality,
   410→    assets,
   411→  })
   412→  track("builder_agent_staged", t.slug)
   413→  return t
   414→}

   416→/** Ask the unified agent for a full site staging — shared by the Builder
   417→ *  button and (via the chat) any other surface that wants a one-shot site. */
   418→export async function askAgentForSite(
   419→  brief: string,
   420→  context: { template?: string | null; business?: string; city?: string } = {},
   421→): Promise<AgentSitePick> {
   422→  const res = await postJson<{ say?: string; actions?: { type: string }[]; error?: string }>("/api/ai/agent", {
   423→    messages: [{ role: "user", content: brief }],
   424→    context,
   425→    hint: "site",
   426→  })
   427→  if (res.error) throw new Error(res.error)
   428→  const site = (res.actions ?? []).find((a) => a.type === "site") as AgentSitePick | undefined
   429→  if (!site) throw new Error("The agent could not pick a starting point — try describing the business in one sentence.")
   430→  return site
   431→}

   433→/** The reference-analysis entry point (the openmontage pattern): paste
   434→ * any site URL, the Page Reader extracts it, the chat AI distills a
   435→ * one-line brief, and the recommendations re-rank against it. */
   436→async function analyzeReference(url: string) {
   437→  if (!url.trim() || state.refBusy) return
   438→  track("ai_skill_used", "builder/reference")
   439→  setState({ refBusy: true, refError: null, refNote: null })
   440→  try {
   441→    const read = await postJson<{ text?: string; title?: string; error?: string }>("/api/ai/reader", { url })
   442→    const text = (read.text ?? "").slice(0, 5000)
   443→    if (!text) throw new Error(read.error ?? "That page returned no readable content.")
   444→    const res = await postJson<{ reply?: string; error?: string }>("/api/ai/chat", {
   445→      system:
   446→        "You turn a website's readable text into a one-line business brief for a website builder. " +
   447→        "Reply with EXACTLY one line, max 20 words, in the shape: 'a <what they do> in <city if known> that <conversion goal>'. " +
   448→        "No quotes, no preamble, no explanation — just the line.",
   449→      messages: [{ role: "user", content: `Website: ${read.title ?? url}\n\nPAGE TEXT:\n${text}` }],
   450→    })
   451→    const line = (res.reply ?? "").replace(/^["']|["']$/g, "").trim()
   452→    if (!line) throw new Error(res.error ?? "Could not distill that page.")
   453→    setState({ brief: line, refBusy: false, refNote: `From ${read.title ?? url} — edit freely, then pick a starting point.` })
   454→  } catch (e) {
   455→    setState({ refBusy: false, refError: e instanceof Error ? e.message : "Could not read that reference." })
   456→  }
   457→}

   459→async function writeCopy() {
   460→  if (state.copyBusy) return
   461→  const t = state.tplSlug ? findTemplate(state.tplSlug) : null
   462→  if (!t) return
   463→  track("ai_skill_used", "builder/copy")
   464→  setState({ copyBusy: true, copyError: null })
   465→  const business = state.name.trim() || t.name
   466→  const city = state.city.trim() || t.city
   467→  const pitch = state.pitch.trim()
   468→  const persona = PERSONAS.find((p) => p.id === state.persona) ?? PERSONAS[0]
   469→  try {
   470→    const res = await postJson<{ reply?: string; error?: string }>("/api/ai/chat", {
   471→      system:
   472→        `You write launch copy for small-business websites. ${persona.style}\n` +
   473→        `Grounded context about the reference template:\n${templateDigest(t)}\n` +
   474→        `The client's business: name "${business}", city ${city}${pitch ? `, pitch: ${pitch}` : ""}.\n` +
   475→        `Write fresh launch copy for THE CLIENT'S business (not the template's demo brand). Keep it concrete and honest, no hype words like "unlock" or "elevate".\n` +
   476→        `Answer with EXACTLY four lines, each starting with the label, nothing else:\n` +
   477→        `HEADLINE: <max 8 words>\nTAGLINE: <one sentence>\nCTA: <max 4 words, a button label>\nABOUT: <2 sentences>`,
   478→      messages: [{ role: "user", content: `Write the launch copy for ${business} in ${city}.` }],
   479→    })
   480→    const raw = res.reply ?? ""
   481→    if (!raw) throw new Error(res.error ?? "No copy came back.")
   482→    const pick = (label: string) => {
   483→      const m = raw.match(new RegExp(`^\\s*${label}\\s*:\\s*(.+)$`, "im"))
   484→      return m ? m[1].trim() : ""
   485→    }
   486→    const block: CopyBlock = {
   487→      headline: pick("HEADLINE"),
   488→      tagline: pick("TAGLINE"),
   489→      cta: pick("CTA"),
   490→      about: pick("ABOUT"),
   491→    }
   492→    if (!block.headline && !block.tagline && !block.about) {
   493→      /* model ignored the format — keep the raw text visible, still useful */
   494→      setState({ copyRaw: raw, copy: null, copyBusy: false })
   495→      return
   496→    }
   497→    setState({ copy: block, copyRaw: raw, copyBusy: false })
   498→    /* re-seed untouched asset prompts with the client's OWN business —
   499→       the template's demo brand never survives into the art step */
   500→    const pos = state.pitch.trim() || t.positioning.split("—")[0].trim()
   501→    const assets = { ...state.assets }
   502→    if (assets.hero.status === "idle" && !assets.hero.touched)
   503→      assets.hero = { ...assets.hero, prompt: `${pos}, ${city}, hero website background, soft natural light, photographic, no text` }
   504→    if (assets.mark.status === "idle" && !assets.mark.touched)
   505→      assets.mark = { ...assets.mark, prompt: `minimal monogram for a business called "${business}", premium, flat vector, centered on plain background, no text effects` }
   506→    if (assets.still.status === "idle" && !assets.still.touched)
   507→      assets.still = { ...assets.still, prompt: `editorial detail photograph — ${pos}, ${city}, shallow depth of field, magazine quality` }
   508→    if (assets.reel.status === "idle" && !assets.reel.touched)
   509→      assets.reel = { ...assets.reel, prompt: `cinematic slow camera move, ${pos.toLowerCase()}, ${city}, warm light, premium mood, no text overlays` }
   510→    setState({ assets })
   511→  } catch (e) {
   512→    setState({ copyBusy: false, copyError: e instanceof Error ? e.message : "The copy service is briefly unavailable." })
   513→  }
   514→}
