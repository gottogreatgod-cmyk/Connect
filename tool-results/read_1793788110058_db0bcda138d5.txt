  1003→function LeadForm() {
  1004→  const [form, setForm] = useState<LeadState>({ name: "", email: "", business: "", industry: "", budget: "", brief: "" })
  1005→  const [errors, setErrors] = useState<{ name?: string; email?: string }>({})
  1006→  const [status, setStatus] = useState<"idle" | "sending" | "done" | "error">("idle")
  1007→  const [refCode, setRefCode] = useState("")
  1008→  const started = useRef(false)
  1009→
  1010→  const bind = (key: keyof LeadState) => ({
  1011→    value: form[key],
  1012→    onChange: (e: ChangeEvent<HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement>) =>
  1013→      setForm((f) => ({ ...f, [key]: e.target.value })),
  1014→  })
  1015→
  1016→  const onFirstFocus = () => {
  1017→    if (!started.current) {
  1018→      started.current = true
  1019→      track("form_started", { source: "forge" })
  1020→    }
  1021→  }
  1022→
  1023→  const onSubmit = async (e: FormEvent<HTMLFormElement>) => {
  1024→    e.preventDefault()
  1025→    const errs: typeof errors = {}
  1026→    if (!form.name.trim()) errs.name = "Your name is required."
  1027→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(form.email.trim())) errs.email = "A complete email, so we can reply."
  1028→    setErrors(errs)
  1029→    if (errs.name || errs.email) {
  1030→      document.getElementById(errs.name ? "forge-name" : "forge-email")?.focus()
  1031→      return
  1032→    }
  1033→    setStatus("sending")
  1034→    try {
  1035→      const res = await fetch("/api/leads", {
  1036→        method: "POST",

  1038→        body: JSON.stringify({ ...form, name: form.name.trim(), email: form.email.trim(), source: "forge" }),
  1039→      })

  1063→          Browse while you wait <ArrowRight className="size-4" aria-hidden />
  1064→        </button>
  1065→      </div>
  1066→    )
  1067→  }
  1068→
  1069→  return (
  1070→    <form
  1071→      noValidate
  1072→      onSubmit={onSubmit}
  1073→      onFocus={onFirstFocus}
  1074→      aria-label="Send a brief"
