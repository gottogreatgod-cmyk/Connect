    52→/* ── keyword inference ────────────────────────────────────────────── */
    53→
    54→const RULES: [ArchetypeId, RegExp][] = [
    55→  ["booking", /book|reserv|appointment|schedul|calendar|availab|seat|sitting|waitlist/i],
    56→  ["portal", /portal|login|sign.?in|account|member|dashboard|patient area|client area/i],
    57→  ["pricing", /pricing|plan|package|subscription|quote builder/i],
    58→  ["reviews", /review|testimonial|praise|rating|star/i],
    59→  ["faq", /faq|question|answer/i],
    60→  ["gallery", /gallery|photo|image|carousel|slider|lightbox|masonry/i],
    61→  ["stats", /stat|metric|number|counter|score|kpi/i],
    62→  ["logos", /logo|trust|partner|client|press|accredit|badge/i],
    63→  ["timeline", /timeline|process|step|how it works|journey|stage/i],
    64→  ["team", /team|staff|profile|people|bio|about/i],
    65→  ["menu", /menu|dish|course|price list|service list|treatment|offering/i],
    66→  ["contact", /contact|location|map|address|hour|direction|footer/i],
    67→  ["cta", /cta|call.?to.?action|closing|announcement|conversion|sticky/i],
    68→  ["hero", /hero|banner|spotlight|headline|landing|masthead/i],
    69→  ["features", /feature|benefit|value|differentiator|explainer|why|included|capabilit/i],
    70→]
    71→
    72→export function inferArchetype(m: Pick<PlaybookModule, "name" | "features" | "section" | "purpose">): ArchetypeId {
    73→  const hay = `${m.name} ${m.section} ${m.features.join(" ")} ${m.purpose}`
    74→  for (const [id, re] of RULES) if (re.test(hay)) return id
    75→  return "spec"
    76→}

   772→
   773→export function ModuleCard({ m, tpl, slot }: { m: PlaybookModule; tpl: TemplateDef; slot: SlotId }) {
   774→  const pack = PACKS[tpl.industry]
   775→  const arch = inferArchetype(m)
   776→  const meta = ARCHETYPES[arch]
   777→  const props = { pack, m }
   778→  /* the factory kit wins: when the playbook id maps to a real, interactive
   779→     kit component (src/modules/kit), mount that with honest demo props
   780→     instead of the generic archetype renderer. */
   781→  const real = hasKitModule(m.id) ? renderKitModule(m, tpl) : null
   782→  return (
   783→    <div data-mod-card data-arch={real ? "kit" : arch} data-slot={slot} style={{ width: "100%", position: "relative", paddingTop: 26 }}>
   784→      <span
   785→        style={{
   786→          position: "absolute",
   787→          top: 0,
   788→          left: 0,
   789→          zIndex: 5,
   790→          background: "var(--primary)",
   791→          color: "var(--on-primary)",
   792→          fontFamily: "var(--mono, monospace)",
   793→          fontSize: 9,
   794→          letterSpacing: "0.14em",
   795→          textTransform: "uppercase",
   796→          padding: "5px 10px",
   797→          borderRadius: "var(--r)",
   798→        }}
   799→        aria-hidden
   800→      >
   801→        {real ? `FACTORY KIT · REAL · ${m.name}` : `${meta.label} · ${m.name}`}
   802→      </span>
   803→      {real ? (
   804→        real
   805→      ) : (
   806→        <>
   807→          {arch === "hero" && <Hero {...props} />}
   808→          {arch === "features" && <Features {...props} />}
   809→          {arch === "menu" && <Menu {...props} />}
   810→          {arch === "pricing" && <Pricing {...props} />}
   811→          {arch === "reviews" && <Reviews {...props} />}
   812→          {arch === "faq" && <Faq {...props} />}
   813→          {arch === "gallery" && <Gallery {...props} />}
   814→          {arch === "stats" && <Stats pack={pack} />}
   815→          {arch === "booking" && <Booking {...props} />}
   816→          {arch === "logos" && <Logos {...props} />}
   817→          {arch === "timeline" && <Timeline {...props} />}
   818→          {arch === "team" && <Team {...props} />}
   819→          {arch === "contact" && <Contact {...props} />}
   820→          {arch === "cta" && <CtaBand {...props} />}
   821→          {arch === "portal" && <Portal {...props} />}
   822→          {arch === "spec" && <SpecCard m={m} tpl={tpl} />}
   823→        </>
   824→      )}
   825→    </div>
   826→  )
   827→}
