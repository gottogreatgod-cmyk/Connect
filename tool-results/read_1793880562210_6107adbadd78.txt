   153→function persistMessages(msgs: AgentMsg[]) {
   154→  if (typeof window === "undefined" || typeof sessionStorage === "undefined") return;
   155→  try {
   156→    const light = msgs.map((m) => ({
   157→      role: m.role,
   158→      text: m.text.slice(0, 4000),
   159→      /* site actions re-stage on restore (small, idempotent); render
   160→         jobs come back as descriptors — never silently re-run */
   161→      actions: m.actions?.filter((a) => a.type === "site"),
   162→      restoredActions: m.actions
   163→        ?.filter((a) => a.type !== "site")
   164→        .map((a) => ({ action: a, restored: true })),
   165→      trace: m.trace?.slice(0, 10),
   166→      memory: m.memory?.slice(0, 8),
   167→    }));
   168→    sessionStorage.setItem(CHAT_KEY, JSON.stringify(light));

   170→    /* quota or storage blocked — the in-memory conversation still works */
   171→  }

   330→/** run-once launcher with cached results: writes to RESULTS always and
   331→ *  to whichever component instance is currently mounted for the rid —
   332→ *  so a render started before navigation keeps updating the view that
   333→ *  remounts after it. */
   334→function useLaunched(ridv: string, run: (write: (r: ActionResult) => void) => Promise<void>): ActionResult {
   335→  const [state, setState] = useState<ActionResult>(() => RESULTS.get(ridv) ?? { phase: "running" });
   336→
   337→  useEffect(() => {
   338→    /* this instance is now the live writer target for the rid */
   339→    LIVE.set(ridv, setState);
   340→    if (!EXECUTED.has(ridv)) {
   341→      EXECUTED.add(ridv);
   342→      void run((r) => {
   343→        RESULTS.set(ridv, r);
   344→        LIVE.get(ridv)?.(r);
   345→      });
   346→    } else {
   347→      /* remount of a running/finished action — push the latest cached
   348→         value through the subscription path (async, never a sync
   349→         setState in the effect body); identical refs bail out in React,
   350→         and later updates flow through LIVE as the job keeps writing */
   351→      const cached = RESULTS.get(ridv);
   352→      if (cached) queueMicrotask(() => LIVE.get(ridv)?.(cached));
   353→    }
   354→    return () => {
   355→      if (LIVE.get(ridv) === setState) LIVE.delete(ridv);
   356→    };
   357→  }, [ridv, run]);
   358→
   359→  return state;
   360→}

   488→function ImageRunner({ action }: { action: { rid: string; prompt: string; size: string } }) {
   489→  const state = useLaunched(action.rid, async (write) => {
   490→    write({ phase: "running", stage: "queued" });
   491→    try {
   492→      const { jobId } = await postJson<{ jobId: string }>("/api/ai/image", { prompt: action.prompt, size: action.size });
   493→      await pollJob("image", jobId, (s: PollState) =>
   494→        write(
   495→          s.status === "done"
   496→            ? { phase: "done", image: s.result?.image }
   497→            : s.status === "error"
   498→              ? { phase: "error", error: s.error ?? "The image render failed." }
   499→              : { phase: "running", stage: s.stage, elapsed: s.elapsed },
   500→        ),
   501→      );
   502→    } catch (e) {
   503→      write({ phase: "error", error: e instanceof Error ? e.message : "The image render failed." });
   504→    }
   505→  });
   506→
   507→  return (
   508→    <ActionShell icon={ImageIcon} title="Image generation" note={action.prompt}>

   687→ *  reload (images are megabytes of base64), so we show what ran instead
   688→ *  of silently firing the job again. */
   689→const RESTORED_ICONS: Record<string, typeof ImageIcon> = {

   697→function RestoredShell({ action }: { action: AgentAction }) {
   698→  const Icon = RESTORED_ICONS[action.type] ?? ImageIcon;
   699→  const note =
   700→    action.type === "image" || action.type === "video"
   701→      ? "prompt"
   702→      : action.type === "speak"
   703→        ? "script"
   704→        : action.type === "search"
   705→          ? "query"
   706→          : "url";
   707→  const detail =
   708→    action.type === "image" || action.type === "video"
   709→      ? action.prompt
   710→      : action.type === "speak"
   711→        ? action.text
   712→        : action.type === "search"
   713→          ? action.query
   714→          : action.type === "read"
   715→            ? action.url
   716→            : "";
   717→  return (
   718→    <ActionShell icon={Icon} title={`${action.type} — ran in an earlier session`} note={`${note}: ${detail}`}>
   719→      <p className="text-[12px] leading-relaxed text-muted-foreground">
   720→        Render results don&apos;t survive a reload — ask again and it re-runs in seconds. The engine&apos;s memory of your
   721→        business is untouched.
   722→      </p>
   723→    </ActionShell>
   724→  );

   736→export function AgentChatPath({
   737→  tpl,
   738→  build,
   739→}: {
   740→  tpl: TemplateDef | null;
   741→  build: { slug: string | null; business: string; city: string; personality: string | null; brief: string };
   742→}) {
   743→  const [messages, setMessagesState] = useState<AgentMsg[]>(() => savedMessages);
   744→  const [input, setInput] = useState("");
   745→  const [thinking, setThinking] = useState(false);
   746→  const [error, setError] = useState<string | null>(null);
   747→  const scrollRef = useRef<HTMLDivElement>(null);
   748→
   749→  const setMessages = (next: AgentMsg[]) => {
   750→    savedMessages = next;
   751→    persistMessages(next);
   752→    setMessagesState(next);
   753→  };
   754→
   755→  /* the live context rides every turn — this is what makes follow-ups
   756→     like "now make a hero image for it" just work */
   757→  const context = useMemo(
   758→    () => ({
   759→      template: build.slug ?? tpl?.slug ?? null,
   760→      business: build.business,
   761→      city: build.city,
   762→      personality: build.personality,
   763→      brief: build.brief,
   764→    }),
   765→    [build.slug, build.business, build.city, build.personality, build.brief, tpl?.slug],
   766→  );
   767→
   768→  useEffect(() => {
   769→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: "smooth" });
   770→  }, [messages, thinking]);
   771→
   772→  const send = async (raw: string) => {
   773→    const text = raw.trim();
   774→    if (!text || thinking) return;
   775→    setInput("");
   776→    setError(null);
   777→    const history: AgentMsg[] = [...savedMessages, { role: "user", text }];
   778→    setMessages(history);
   779→    setThinking(true);
   780→    track("agent_chat_used");
   781→    try {
   782→      const res = await postJson<{
   783→        say?: string;
   784→        actions?: IncomingAction[];
   785→        trace?: TraceStep[];
   786→        memory?: string[];
   787→        error?: string;
   788→      }>("/api/ai/agent", {
   789→        messages: history.slice(-8).map((m) => ({ role: m.role, content: m.text })),
   790→        context,
   791→      });
   792→      if (res.error) throw new Error(res.error);
   793→      const actions = (res.actions ?? []).map((a) => ({ ...a, rid: rid() }) as AgentAction);
   794→      setMessages([
   795→        ...history,
   796→        {
   797→          role: "assistant",
   798→          text: res.say ?? "Done.",
   799→          actions,
   800→          trace: Array.isArray(res.trace) ? res.trace.slice(0, 10) : undefined,
   801→          memory: Array.isArray(res.memory) ? res.memory : undefined,
   802→        },
   803→      ]);
   804→    } catch (e) {
   805→      setError(e instanceof Error ? e.message : "The agent is briefly unavailable.");
   806→    } finally {
   807→      setThinking(false);
   808→    }
   809→  };
   810→
   811→  return (
   812→    <div className="flex min-h-[34rem] flex-1 flex-col gap-3">
   813→      <div ref={scrollRef} className="min-h-64 flex-1 space-y-4 overflow-y-auto rounded-xl border border-border bg-muted/20 p-4">
   814→        {messages.map((m, i) => (
   815→          <div key={i} className={m.role === "user" ? "ml-auto max-w-[85%]" : "mr-auto max-w-[92%] space-y-2.5"}>
   816→            <div
   817→              className={
   818→                m.role === "user"
   819→                  ? "rounded-xl rounded-br-sm bg-primary px-3.5 py-2.5 text-sm leading-relaxed text-primary-foreground shadow-sm"
   820→                  : "whitespace-pre-wrap rounded-xl rounded-bl-sm border border-border bg-card px-3.5 py-2.5 text-sm leading-relaxed shadow-sm"
   821→              }
   822→            >
   823→              {m.text}
   824→            </div>
   825→            {m.actions?.map((a) => (
   826→              <ActionRunner key={a.rid} action={a} />
   827→            ))}
   828→            {m.restoredActions?.map((r, i) => (
   829→              <RestoredShell key={`restored-${i}`} action={r.action} />
   830→            ))}
   831→            {(m.trace?.length || m.memory?.length) && <EngineTrace trace={m.trace} memory={m.memory} />}
   832→          </div>
   833→        ))}
   834→        {thinking && (
   835→          <div className="mr-auto w-[92%]">
   836→            <EngineThinking />
   837→          </div>
   838→        )}
   839→      </div>
   840→      {error && <ErrorNote>{error}</ErrorNote>}
   841→      <div className="flex flex-wrap items-center gap-2">
   842→        {messages.length <= 1 ? (
   843→          CHIPS.map((c) => (
   844→            <button key={c} type="button" className={`${btnGhostCx} h-8 px-3 text-xs`} onClick={() => void send(c)}>
   845→              {c}
   846→            </button>
   847→          ))
   848→        ) : (
   849→          <button
   850→            type="button"
   851→            className={`${btnGhostCx} h-8 px-3 text-xs`}
   852→            onClick={() => {
   853→              setMessages([HELLO])
   854→              setError(null)
   855→            }}
   856→          >
   857→            <Square className="size-3" aria-hidden /> New chat
   858→          </button>
   859→        )}
   860→      </div>
   861→      <form
   862→        className="flex gap-2"
   863→        onSubmit={(e) => {
   864→          e.preventDefault();
   865→          void send(input);
   866→        }}
   867→      >
   868→        <input
   869→          className={inputCx}
   870→          value={input}
   871→          onChange={(e) => setInput(e.target.value)}
   872→          placeholder='Ask for anything — "build a site for my bakery, then make the hero image"'
   873→          aria-label="Message the agent"
   874→        />
   875→        <button type="submit" className={btnPrimaryCx} disabled={thinking || !input.trim()} aria-label="Send message">
   876→          <Send className="size-4" aria-hidden />
   877→        </button>
   878→      </form>
   879→      <p className="flex flex-wrap items-center gap-x-2 gap-y-1 text-[11px] text-muted-foreground">
   880→        <Sparkles className="size-3.5" aria-hidden />
   881→        one engine, every skill — planner → parallel sub-agents → verification → memory · site builds auto-stage with a live preview · the conversation survives navigation
   882→      </p>
   883→    </div>
   884→  );
   885→}
